cmake_minimum_required(VERSION 3.20)
project(modelconnect_showcase_vision LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_WINDOWS_EXPORT_ALL_SYMBOLS ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(TRTMC_SOURCE_DIR "" CACHE PATH "ModelConnect source checkout")
set(TRTMC_BUILD_BACKEND_TRT OFF CACHE BOOL "" FORCE)
set(TRTMC_BUILD_BACKEND_RTX ON CACHE BOOL "" FORCE)
set(TRTMC_FAMILIES sam CACHE STRING "" FORCE)
set(TRTMC_BUILD_CLI OFF CACHE BOOL "" FORCE)
set(TRTMC_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(TRTMC_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(TRTMC_BUILD_WINDOWS_VOICECHAT OFF CACHE BOOL "" FORCE)
set(TRTMC_ENABLE_BYOK OFF CACHE BOOL "" FORCE)
add_subdirectory("${TRTMC_SOURCE_DIR}" "${CMAKE_BINARY_DIR}/trtmc")
find_package(nlohmann_json 3.11 REQUIRED)
add_executable(trtmc_vision_bridge main.cpp)
target_include_directories(trtmc_vision_bridge PRIVATE "${TRTMC_SOURCE_DIR}")
target_compile_options(trtmc_vision_bridge PRIVATE /W4 /permissive- /utf-8)
target_compile_definitions(trtmc_vision_bridge PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(trtmc_vision_bridge PRIVATE trtmc_runtime trtmc_core nlohmann_json::nlohmann_json)
add_dependencies(trtmc_vision_bridge trtmc_backend_rtx trtmc_model_sam)
install(TARGETS trtmc_vision_bridge RUNTIME DESTINATION bin)
add_executable(test_vision_contract test_vision_contract.cpp)
target_include_directories(test_vision_contract PRIVATE "${TRTMC_SOURCE_DIR}/core/runtime/include")
target_compile_options(test_vision_contract PRIVATE /W4 /permissive- /utf-8)
target_compile_definitions(test_vision_contract PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(test_vision_contract PRIVATE nlohmann_json::nlohmann_json)
