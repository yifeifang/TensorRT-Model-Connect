cmake_minimum_required(VERSION 3.20)
project(modelconnect_showcase_geometry LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_WINDOWS_EXPORT_ALL_SYMBOLS ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(TRTMC_SOURCE_DIR "" CACHE PATH "ModelConnect source checkout")
set(TRTMC_BUILD_BACKEND_TRT OFF CACHE BOOL "" FORCE)
set(TRTMC_BUILD_BACKEND_RTX ON CACHE BOOL "" FORCE)
set(TRTMC_FAMILIES moge CACHE STRING "" FORCE)
set(TRTMC_BUILD_CLI OFF CACHE BOOL "" FORCE)
set(TRTMC_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(TRTMC_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(TRTMC_BUILD_WINDOWS_VOICECHAT OFF CACHE BOOL "" FORCE)
set(TRTMC_ENABLE_BYOK OFF CACHE BOOL "" FORCE)
add_subdirectory("${TRTMC_SOURCE_DIR}" "${CMAKE_BINARY_DIR}/trtmc")
if(WIN32)
  set_property(TARGET trtmc_model_moge PROPERTY COMPILE_OPTIONS /W4 /permissive- /utf-8)
  install(TARGETS trtmc_model_moge RUNTIME DESTINATION bin ARCHIVE DESTINATION lib)
endif()
find_package(nlohmann_json 3.11 REQUIRED)
add_executable(trtmc_geometry_bridge main.cpp)
target_include_directories(trtmc_geometry_bridge PRIVATE "${TRTMC_SOURCE_DIR}")
target_compile_options(trtmc_geometry_bridge PRIVATE /W4 /permissive- /utf-8)
target_compile_definitions(trtmc_geometry_bridge PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(trtmc_geometry_bridge PRIVATE trtmc_runtime trtmc_core nlohmann_json::nlohmann_json)
add_dependencies(trtmc_geometry_bridge trtmc_backend_rtx trtmc_model_moge)
install(TARGETS trtmc_geometry_bridge RUNTIME DESTINATION bin)
add_executable(geometry_engine_probe engine_probe.cpp)
target_include_directories(geometry_engine_probe PRIVATE "${TRTMC_SOURCE_DIR}" "${TRTMC_CUDA_INCLUDE_DIR}")
target_compile_options(geometry_engine_probe PRIVATE /W4 /permissive- /utf-8)
target_compile_definitions(geometry_engine_probe PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(geometry_engine_probe PRIVATE trtmc_core trtmc_backend_rtx nlohmann_json::nlohmann_json)
install(TARGETS geometry_engine_probe RUNTIME DESTINATION bin)
