cmake_minimum_required(VERSION 3.20)
project(modelconnect_showcase_embedding LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_WINDOWS_EXPORT_ALL_SYMBOLS ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(TRTMC_SOURCE_DIR "" CACHE PATH "ModelConnect source checkout")
set(TRTMC_BUILD_BACKEND_TRT OFF CACHE BOOL "" FORCE)
set(TRTMC_BUILD_BACKEND_RTX ON CACHE BOOL "" FORCE)
set(TRTMC_FAMILIES bert CACHE STRING "" FORCE)
set(TRTMC_BUILD_CLI OFF CACHE BOOL "" FORCE)
set(TRTMC_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(TRTMC_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(TRTMC_BUILD_WINDOWS_VOICECHAT OFF CACHE BOOL "" FORCE)
set(TRTMC_ENABLE_BYOK OFF CACHE BOOL "" FORCE)
add_subdirectory("${TRTMC_SOURCE_DIR}" "${CMAKE_BINARY_DIR}/trtmc")
if(WIN32)
  include("${CMAKE_CURRENT_SOURCE_DIR}/pad-static-inputs.cmake")
  get_target_property(_bert_sources trtmc_model_bert SOURCES)
  set(_windows_bert_sources "${CMAKE_CURRENT_SOURCE_DIR}/single_device.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/unicode_normalize.cpp" "${_padded_pipeline}" "${_normalized_tokenizer}")
  foreach(_source IN LISTS _bert_sources)
    if(NOT _source MATCHES "(^|/)(distributed_runtime|pipeline|unigram_tokenizer)\\.cpp$")
      if(IS_ABSOLUTE "${_source}")
        list(APPEND _windows_bert_sources "${_source}")
      else()
        list(APPEND _windows_bert_sources "${TRTMC_SOURCE_DIR}/families/bert/runtime/${_source}")
      endif()
    endif()
  endforeach()
  set_property(TARGET trtmc_model_bert PROPERTY SOURCES "${_windows_bert_sources}")
  set_property(TARGET trtmc_model_bert PROPERTY COMPILE_OPTIONS /W4 /permissive- /utf-8)
  target_include_directories(trtmc_model_bert PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
  target_link_libraries(trtmc_model_bert PRIVATE Normaliz)
  install(TARGETS trtmc_model_bert RUNTIME DESTINATION bin ARCHIVE DESTINATION lib)
endif()
find_package(nlohmann_json 3.11 REQUIRED)
add_executable(trtmc_embedding_bridge main.cpp)
target_include_directories(trtmc_embedding_bridge PRIVATE "${TRTMC_SOURCE_DIR}")
target_compile_options(trtmc_embedding_bridge PRIVATE /W4 /permissive- /utf-8)
target_compile_definitions(trtmc_embedding_bridge PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(trtmc_embedding_bridge PRIVATE trtmc_runtime trtmc_core trtmc_model_bert nlohmann_json::nlohmann_json)
add_dependencies(trtmc_embedding_bridge trtmc_backend_rtx)
install(TARGETS trtmc_embedding_bridge RUNTIME DESTINATION bin)
