cmake_minimum_required(VERSION 3.20)
project(modelconnect_showcase_understanding LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_WINDOWS_EXPORT_ALL_SYMBOLS ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(TRTMC_SOURCE_DIR "" CACHE PATH "ModelConnect source checkout")
set(TRTMC_BUILD_BACKEND_TRT OFF CACHE BOOL "" FORCE)
set(TRTMC_BUILD_BACKEND_RTX ON CACHE BOOL "" FORCE)
set(TRTMC_FAMILIES qwen_vl CACHE STRING "" FORCE)
set(TRTMC_BUILD_CLI OFF CACHE BOOL "" FORCE)
set(TRTMC_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(TRTMC_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(TRTMC_BUILD_WINDOWS_VOICECHAT OFF CACHE BOOL "" FORCE)
set(TRTMC_ENABLE_BYOK OFF CACHE BOOL "" FORCE)
add_subdirectory("${TRTMC_SOURCE_DIR}" "${CMAKE_BINARY_DIR}/trtmc")
if(WIN32)
  # Single-device portability changes are scoped to this build target. Shared
  # vendor sources retain their original distributed implementation and flags.
  get_target_property(_sources trtmc_model_qwen_vl SOURCES)
  set(_windows_sources "${CMAKE_CURRENT_SOURCE_DIR}/single_device.cpp")
  foreach(_source IN LISTS _sources)
    if(NOT _source MATCHES "(^|/)distributed_runtime\\.cpp$")
      if(IS_ABSOLUTE "${_source}")
        list(APPEND _windows_sources "${_source}")
      else()
        list(APPEND _windows_sources "${TRTMC_SOURCE_DIR}/families/qwen_vl/runtime/${_source}")
      endif()
    endif()
  endforeach()
  set_property(TARGET trtmc_model_qwen_vl PROPERTY SOURCES "${_windows_sources}")
  set_property(TARGET trtmc_model_qwen_vl PROPERTY COMPILE_OPTIONS /W4 /permissive- /utf-8)
  set_source_files_properties("${TRTMC_SOURCE_DIR}/families/qwen_vl/runtime/image_preprocessor.cpp"
    DIRECTORY "${TRTMC_SOURCE_DIR}/families/qwen_vl/runtime" PROPERTIES COMPILE_OPTIONS "/W4;/utf-8")
  target_compile_definitions(trtmc_model_qwen_vl PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  install(TARGETS trtmc_model_qwen_vl RUNTIME DESTINATION bin ARCHIVE DESTINATION lib)
endif()
find_package(nlohmann_json 3.11 REQUIRED)
add_executable(trtmc_understanding_bridge main.cpp)
target_include_directories(trtmc_understanding_bridge PRIVATE "${TRTMC_SOURCE_DIR}")
target_compile_options(trtmc_understanding_bridge PRIVATE /W4 /permissive- /utf-8)
target_compile_definitions(trtmc_understanding_bridge PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(trtmc_understanding_bridge PRIVATE trtmc_runtime trtmc_core trtmc_model_qwen_vl nlohmann_json::nlohmann_json)
add_dependencies(trtmc_understanding_bridge trtmc_backend_rtx trtmc_model_qwen_vl)
install(TARGETS trtmc_understanding_bridge RUNTIME DESTINATION bin)
add_executable(test_understanding_contract test_understanding_contract.cpp)
target_compile_options(test_understanding_contract PRIVATE /W4 /permissive- /utf-8)
target_compile_definitions(test_understanding_contract PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(test_understanding_contract PRIVATE nlohmann_json::nlohmann_json)
