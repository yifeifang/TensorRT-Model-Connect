cmake_minimum_required(VERSION 3.20)
project(modelconnect_showcase_asr LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_WINDOWS_EXPORT_ALL_SYMBOLS ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(TRTMC_SOURCE_DIR "" CACHE PATH "Isolated ModelConnect source checkout")
set(TRTMC_BUILD_BACKEND_TRT OFF CACHE BOOL "" FORCE)
set(TRTMC_BUILD_BACKEND_RTX ON CACHE BOOL "" FORCE)
set(TRTMC_FAMILIES whisper CACHE STRING "" FORCE)
set(TRTMC_BUILD_CLI OFF CACHE BOOL "" FORCE)
set(TRTMC_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(TRTMC_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(TRTMC_BUILD_WINDOWS_VOICECHAT OFF CACHE BOOL "" FORCE)
set(TRTMC_ENABLE_BYOK OFF CACHE BOOL "" FORCE)
add_subdirectory("${TRTMC_SOURCE_DIR}" "${CMAKE_BINARY_DIR}/trtmc")
find_package(nlohmann_json 3.11 REQUIRED)
add_executable(trtmc_asr_bridge main.cpp)
target_include_directories(trtmc_asr_bridge PRIVATE "${TRTMC_SOURCE_DIR}")
target_compile_options(trtmc_asr_bridge PRIVATE /W4 /permissive- /utf-8)
target_compile_definitions(trtmc_asr_bridge PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(trtmc_asr_bridge PRIVATE trtmc_runtime trtmc_core nlohmann_json::nlohmann_json)
add_dependencies(trtmc_asr_bridge trtmc_backend_rtx trtmc_model_whisper)
install(TARGETS trtmc_asr_bridge RUNTIME DESTINATION bin)
add_executable(test_showcase_asr_wav test_wav.cpp)
target_compile_options(test_showcase_asr_wav PRIVATE /W4 /permissive- /utf-8)
