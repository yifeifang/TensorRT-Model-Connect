cmake_minimum_required(VERSION 3.20)
project(modelconnect_showcase_text LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_WINDOWS_EXPORT_ALL_SYMBOLS ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(TRTMC_SOURCE_DIR "" CACHE PATH "ModelConnect source checkout")
set(TRTMC_BUILD_BACKEND_TRT OFF CACHE BOOL "" FORCE)
set(TRTMC_BUILD_BACKEND_RTX ON CACHE BOOL "" FORCE)
set(TRTMC_FAMILIES qwen CACHE STRING "" FORCE)
set(TRTMC_BUILD_CLI OFF CACHE BOOL "" FORCE)
set(TRTMC_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(TRTMC_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(TRTMC_BUILD_WINDOWS_VOICECHAT OFF CACHE BOOL "" FORCE)
set(TRTMC_ENABLE_BYOK OFF CACHE BOOL "" FORCE)
add_subdirectory("${TRTMC_SOURCE_DIR}" "${CMAKE_BINARY_DIR}/trtmc")
find_package(nlohmann_json 3.11 REQUIRED)
add_executable(trtmc_text_bridge main.cpp)
target_include_directories(trtmc_text_bridge PRIVATE "${TRTMC_SOURCE_DIR}")
target_compile_options(trtmc_text_bridge PRIVATE /W4 /permissive- /utf-8)
target_compile_definitions(trtmc_text_bridge PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(trtmc_text_bridge PRIVATE trtmc_runtime trtmc_core nlohmann_json::nlohmann_json)
add_dependencies(trtmc_text_bridge trtmc_backend_rtx trtmc_model_qwen)
install(TARGETS trtmc_text_bridge RUNTIME DESTINATION bin)
add_executable(test_showcase_chat_template test_chat_template.cpp)
target_include_directories(test_showcase_chat_template PRIVATE "${TRTMC_SOURCE_DIR}")
target_compile_options(test_showcase_chat_template PRIVATE /W4 /permissive- /utf-8)
target_link_libraries(test_showcase_chat_template PRIVATE trtmc_model_qwen nlohmann_json::nlohmann_json)
set_target_properties(test_showcase_chat_template PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/trtmc")
